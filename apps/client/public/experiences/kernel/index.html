From a raw document to a booked invoice

A workflow is plain code with five steps you can read, test and replay. Every run is logged with its inputs, so a wrong answer can be traced.

invoice-triage.js Running
  1. import { workflow } from "@kernel/sdk";
  3. export default workflow("invoice-triage", async (run) => {
  4. const doc = await run.ingest({ source: "inbox/invoices" });
  5. const facts = await run.retrieve(doc, { index: "vendors", topK: 5 });
  7. const draft = await run.reason({
  8. model: "kern-m",
  9. task: "Match to purchase order and flag mismatches",
  10. context: facts,
  11. schema: { po: "string", total: "number", flags: "string[]" },
  12. });
  14. const approved = await run.review(draft, { when: draft.flags.length > 0 });
  15. return run.ship(approved, { to: "erp.bills" });
  16. });
